/* ======================================================== 台灣房價網站 housing.js ======================================================== 功能： 1. 載入房價 / 租金 / 所得 JSON 2. 載入台灣縣市 TopoJSON 3. 顯示台灣本島地圖 4. 預設台北市 5. 點擊縣市切換資料 6. 滑鼠移入顯示縣市名稱 ======================================================== */ /* ======================================================== DOM ======================================================== */ const tableBody = document.getElementById("tableBody"); const count = document.getElementById("count"); const selectedCity = document.getElementById("selectedCity"); const taiwanMap = document.getElementById("taiwanMap"); const mapTooltip = document.getElementById("mapTooltip"); /* ======================================================== 全域資料 ======================================================== */ let housingData = []; let selectedCityName = "台北市"; let mapSvg = null; let mapProjection = null; let mapPath = null; /* ======================================================== 城市名稱標準化 TopoJSON 有些名稱使用「臺」 我們網站資料使用「台」 統一成「台」 ======================================================== */ function normalizeCityName(name) { if (!name) { return ""; } return String(name) .replaceAll("臺", "台") .trim(); } /* ======================================================== 格式化數字 ======================================================== */ function formatNumber(value) { if ( value === null || value === undefined || value === "" ) { return "—"; } const number = Number(value); if (!Number.isFinite(number)) { return "—"; } return number.toLocaleString("zh-TW"); } /* ======================================================== 房價 ======================================================== */ function formatPrice(value) { if ( value === null || value === undefined || value === "" ) { return "—"; } return ( formatNumber(value) + " 元/坪" ); } /* ======================================================== 租金 ======================================================== */ function formatRent(value) { if ( value === null || value === undefined || value === "" ) { return "—"; } return ( formatNumber(value) + " 元/月" ); } /* ======================================================== 房價所得比 ======================================================== */ function formatIncomeRatio(value) { if ( value === null || value === undefined || value === "" ) { return "—"; } const number = Number(value); if (!Number.isFinite(number)) { return "—"; } return ( number.toFixed(1) + " 倍" ); } /* ======================================================== 顯示表格 ======================================================== */ function renderTable() { const city = normalizeCityName( selectedCityName ); /* ---------------------------------------------------- 找出目前縣市 ---------------------------------------------------- */ const filteredData = housingData.filter(item => { return ( normalizeCityName(item.city) === city ); }); /* ---------------------------------------------------- 顯示目前縣市 ---------------------------------------------------- */ selectedCity.textContent = city || "台北市"; /* ---------------------------------------------------- 顯示筆數 ---------------------------------------------------- */ count.textContent = `共 ${filteredData.length} 個行政區`; /* ---------------------------------------------------- 沒資料 ---------------------------------------------------- */ if (filteredData.length === 0) { tableBody.innerHTML = ` 目前沒有這個縣市的資料 `; return; } /* ---------------------------------------------------- 排序 ---------------------------------------------------- */ filteredData.sort((a, b) => { return String(a.district) .localeCompare( String(b.district), "zh-TW" ); }); /* ---------------------------------------------------- 建立表格 ---------------------------------------------------- */ tableBody.innerHTML = filteredData .map(item => { return ` ${item.district} ${formatPrice( item.median_price_per_ping )} ${formatRent( item.average_monthly_rent )} ${formatIncomeRatio( item.price_income_ratio )} `; }) .join(""); } /* ======================================================== 選擇縣市 ======================================================== */ function selectCity(cityName) { const normalized = normalizeCityName(cityName); if (!normalized) { return; } selectedCityName = normalized; /* 更新地圖選取狀態 */ if (mapSvg) { mapSvg .selectAll(".county") .classed( "selected", function () { const name = normalizeCityName( d3.select(this) .attr("data-city") ); return name === normalized; } ); } /* 更新表格 */ renderTable(); /* 手機點擊地圖後 自動把畫面移到資料區 */ if ( window.innerWidth <= 650 ) { const table = document.querySelector( ".table-container" ); if (table) { table.scrollIntoView({ behavior: "smooth", block: "start" }); } } } /* ======================================================== 地圖 Tooltip ======================================================== */ function showTooltip( event, cityName ) { if (!mapTooltip) { return; } mapTooltip.innerHTML = ` ${cityName}
點擊查看房價與租金 `; mapTooltip.style.display = "block"; const rect = taiwanMap.getBoundingClientRect(); let x = event.clientX - rect.left + 12; let y = event.clientY - rect.top + 12; /* 防止 Tooltip 超出右側 */ if ( x + 170 > taiwanMap.clientWidth ) { x -= 180; } /* 防止 Tooltip 超出下方 */ if ( y + 70 > taiwanMap.clientHeight ) { y -= 80; } mapTooltip.style.left = `${x}px`; mapTooltip.style.top = `${y}px`; } /* ======================================================== 隱藏 Tooltip ======================================================== */ function hideTooltip() { if (!mapTooltip) { return; } mapTooltip.style.display = "none"; } /* ======================================================== 建立台灣地圖 ======================================================== */ async function createTaiwanMap() { /* ---------------------------------------------------- Taiwan Atlas ---------------------------------------------------- */ const mapUrl = "https://cdn.jsdelivr.net/npm/taiwan-atlas@2021.9.20/counties-10t.json"; const response = await fetch( mapUrl, { cache: "no-store" } ); if (!response.ok) { throw new Error( `地圖載入失敗 HTTP ${response.status}` ); } const topology = await response.json(); /* ---------------------------------------------------- 取得 counties ---------------------------------------------------- */ const allCounties = topojson.feature( topology, topology.objects.counties ); /* ---------------------------------------------------- 排除外島 澎湖 金門 連江 ---------------------------------------------------- */ const excludedCities = new Set([ "澎湖縣", "金門縣", "連江縣" ]); const counties = allCounties.features.filter( feature => { const name = normalizeCityName( feature.properties.COUNTYNAME ); return !excludedCities.has( name ); } ); /* ---------------------------------------------------- 建立 GeoJSON ---------------------------------------------------- */ const geojson = { type: "FeatureCollection", features: counties }; /* ---------------------------------------------------- 清空舊地圖 ---------------------------------------------------- */ taiwanMap .querySelectorAll("svg") .forEach(svg => svg.remove()); /* ---------------------------------------------------- 取得容器尺寸 ---------------------------------------------------- */ const width = taiwanMap.clientWidth; const height = taiwanMap.clientHeight; /* ---------------------------------------------------- 建立 SVG ---------------------------------------------------- */ mapSvg = d3.select("#taiwanMap") .append("svg") .attr( "viewBox", `0 0 ${width} ${height}` ) .attr( "preserveAspectRatio", "xMidYMid meet" ); /* ---------------------------------------------------- 地圖投影 直接依照「台灣本島縣市」自動縮放 ---------------------------------------------------- */ mapProjection = d3.geoMercator(); mapProjection.fitSize( [width, height], geojson ); /* ---------------------------------------------------- Path ---------------------------------------------------- */ mapPath = d3.geoPath() .projection( mapProjection ); /* ---------------------------------------------------- 繪製縣市 ---------------------------------------------------- */ mapSvg .selectAll(".county") .data(counties) .enter() .append("path") .attr( "class", "county" ) .attr( "d", mapPath ) .attr( "data-city", d => normalizeCityName( d.properties.COUNTYNAME ) ) /* ------------------------------------------------ 滑鼠移入 ------------------------------------------------ */ .on( "mouseenter", function(event, d) { const city = normalizeCityName( d.properties.COUNTYNAME ); showTooltip( event, city ); } ) /* ------------------------------------------------ 滑鼠移動 ------------------------------------------------ */ .on( "mousemove", function(event, d) { const city = normalizeCityName( d.properties.COUNTYNAME ); showTooltip( event, city ); } ) /* ------------------------------------------------ 滑鼠離開 ------------------------------------------------ */ .on( "mouseleave", function() { hideTooltip(); } ) /* ------------------------------------------------ 點擊縣市 ------------------------------------------------ */ .on( "click", function(event, d) { const city = normalizeCityName( d.properties.COUNTYNAME ); selectCity( city ); } ); /* ---------------------------------------------------- 預設台北市 ---------------------------------------------------- */ selectCity("台北市"); } /* ======================================================== 載入房價資料 ======================================================== */ async function loadHousingData() { try { const response = await fetch( "data/taiwan_housing.json?v=4", { cache: "no-store" } ); if (!response.ok) { throw new Error( `房價資料 HTTP ${response.status}` ); } housingData = await response.json(); /* ------------------------------------------------ 檢查格式 ------------------------------------------------ */ if ( !Array.isArray( housingData ) ) { throw new Error( "taiwan_housing.json 必須是陣列" ); } /* ------------------------------------------------ 建立地圖 ------------------------------------------------ */ await createTaiwanMap(); /* ------------------------------------------------ 顯示預設資料 ------------------------------------------------ */ renderTable(); } catch (error) { console.error( "網站資料載入失敗：", error ); selectedCity.textContent = "台北市"; count.textContent = ""; tableBody.innerHTML = ` 資料載入失敗，請重新整理頁面 `; } } /* ======================================================== 視窗大小改變 讓地圖重新 fit ======================================================== */ let resizeTimer = null; window.addEventListener( "resize", function() { clearTimeout( resizeTimer ); resizeTimer = setTimeout( async function() { try { await createTaiwanMap(); } catch (error) { console.error( "地圖重新繪製失敗：", error ); } }, 200 ); } ); /* ======================================================== 開始 ======================================================== */ loadHousingData();
